# 多环境配置加载实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 启动时通过位置参数（`python main.py test`）或环境变量（`OPENMANUS_ENV=test`）选择 `config/config_{env}.toml` 配置文件，默认 dev。

**Architecture:** 环境变量 `OPENMANUS_ENV` 是唯一桥接点：`app/config.py` 的 `_get_config_path()` 按环境变量选文件（默认 dev、动态枚举、缺失报错、dev 回退 example）；入口脚本解析位置参数后写入环境变量（参数优先），再延迟 import app 模块（config 单例在首次 import 时加载）。

**Tech Stack:** Python 3.11-3.13、tomllib、argparse（标准库）

## Global Constraints

- 环境优先级：位置参数 > `OPENMANUS_ENV` 环境变量 > 默认 `"dev"`
- 合法环境集合 = config 目录下实际存在的 `config_*.toml` 文件（动态枚举，不做白名单）
- 指定环境文件缺失 → `FileNotFoundError`，报错信息含 config 目录下可用环境清单
- 默认 dev 且 `config_dev.toml` 缺失 → 回退 `config.example.toml`（现有行为保留）
- `OPENMANUS_ENV=""`（空串/空白）→ 视为未设置，走默认 dev
- config 层**不读 sys.argv**，只认环境变量
- 任何 `from app.config import ...` 之前必须先写入 `OPENMANUS_ENV`
- `config_dev.toml` / `config_test.toml` 不入库（config/.gitignore 追加）
- 代码注释简体中文、标识符英文、验收前必须跑测试

```python
"""启动入口公共辅助：多环境位置参数解析与 OPENMANUS_ENV 桥接。

注意：本模块只依赖标准库，import 它不会触发 app.config 加载。
入口脚本必须遵循：先解析位置参数并调用 apply_env_from_args 写入
OPENMANUS_ENV，之后才 import app.* 模块 —— app.config 的模块级单例
在首次 import 时即完成配置加载，环境变量必须在它之前就位。
"""

import argparse
import os
from typing import Any


def add_env_argument(parser: argparse.ArgumentParser) -> None:
    """给入口 parser 添加可选位置参数 env（对应 config/config_{env}.toml）。"""
    parser.add_argument(
        "env",
        nargs="?",
        default=None,
        help="部署环境（config/config_{env}.toml，如 dev/test），默认 dev",
    )


def apply_env_from_args(args: Any) -> None:
    """位置参数优先写入 OPENMANUS_ENV（仅当显式传入非空值）。

    未传或传空串时不覆盖已有环境变量，保留环境变量通道的优先级。
    """
    env = getattr(args, "env", None)
    if env:
        os.environ["OPENMANUS_ENV"] = env
```

- [ ] **Step 4: 改造 `run_mcp.py`**

原理：`MCPRunner` 类的 `__init__` 引用 `config`（模块级名字），类方法引用 `logger`——将类定义整体移入 `run_mcp()` 函数体内后，方法体的名字解析会通过闭包捕获函数局部变量 `config`/`logger`/`MCPAgent`（Python 标准行为：类定义在函数内时方法闭包可见函数局部命名空间）。`parse_args` 保持模块级（不依赖 app 模块）。整个文件改造后为：

```python
#!/usr/bin/env python
import argparse
import asyncio
import sys

from app.utils.entry import add_env_argument, apply_env_from_args


def parse_args() -> argparse.Namespace:
    """Parse command line arguments."""
    parser = argparse.ArgumentParser(description="Run the MCP Agent")
    add_env_argument(parser)
    parser.add_argument(
        "--connection",
        "-c",
        choices=["stdio", "sse"],
        default="stdio",
        help="Connection type: stdio or sse",
    )
    parser.add_argument(
        "--server-url",
        default="http://127.0.0.1:8000/sse",
        help="URL for SSE connection",
    )
    parser.add_argument(
        "--interactive", "-i", action="store_true", help="Run in interactive mode"
    )
    parser.add_argument("--prompt", "-p", help="Single prompt to execute and exit")
    return parser.parse_args()


async def run_mcp() -> None:
    """Main entry point for the MCP runner."""
    # 先解析环境参数并写入 OPENMANUS_ENV，再加载 app 模块（config 单例依赖环境变量就位）
    args = parse_args()
    apply_env_from_args(args)

    from app.agent.mcp import MCPAgent
    from app.config import config
    from app.logger import logger

    class MCPRunner:
        """Runner class for MCP Agent with proper path handling and configuration."""

        def __init__(self):
            self.root_path = config.root_path
            self.server_reference = config.mcp_config.server_reference
            self.agent = MCPAgent()

        async def initialize(
            self,
            connection_type: str,
            server_url: str | None = None,
        ) -> None:
            """Initialize the MCP agent with the appropriate connection."""
            logger.info(f"Initializing MCPAgent with {connection_type} connection...")

            if connection_type == "stdio":
                await self.agent.initialize(
                    connection_type="stdio",
                    command=sys.executable,
                    args=["-m", self.server_reference],
                )
            else:  # sse
                await self.agent.initialize(connection_type="sse", server_url=server_url)

            logger.info(f"Connected to MCP server via {connection_type}")

        async def run_interactive(self) -> None:
            """Run the agent in interactive mode."""
            print("\nMCP Agent Interactive Mode (type 'exit' to quit)\n")
            while True:
                user_input = input("\nEnter your request: ")
                if user_input.lower() in ["exit", "quit", "q"]:
                    break
                response = await self.agent.run(user_input)
                print(f"\nAgent: {response}")

        async def run_single_prompt(self, prompt: str) -> None:
            """Run the agent with a single prompt."""
            await self.agent.run(prompt)

        async def run_default(self) -> None:
            """Run the agent in default mode."""
            prompt = input("Enter your prompt: ")
            if not prompt.strip():
                logger.warning("Empty prompt provided.")
                return

            logger.warning("Processing your request...")
            await self.agent.run(prompt)
            logger.info("Request processing completed.")

        async def cleanup(self) -> None:
            """Clean up agent resources."""
            await self.agent.cleanup()
            logger.info("Session ended")

    runner = MCPRunner()

    try:
        await runner.initialize(args.connection, args.server_url)

        if args.prompt:
            await runner.run_single_prompt(args.prompt)
        elif args.interactive:
            await runner.run_interactive()
        else:
            await runner.run_default()

    except KeyboardInterrupt:
        logger.info("Program interrupted by user")
    except Exception as e:
        logger.error(f"Error running MCPAgent: {str(e)}", exc_info=True)
        sys.exit(1)
    finally:
        await runner.cleanup()


if __name__ == "__main__":
    asyncio.run(run_mcp())
```
